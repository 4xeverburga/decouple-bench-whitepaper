% The whitepaper's shared figure style, after DESIGN.md (ChiffonStack).
%
% A figure is one file under `figures/`: it compiles on its own
% (standalone, so it drops into any document) and is included by
% `atlas.tex`, which exists only to review them side by side.
%
% Everything visual lives here. A figure that invents its own colour,
% arrow or border is a bug: the point of a style file is that figure 4
% looks like it was drawn by whoever drew figure 1.
%
% Built with LuaLaTeX: the type is Quicksand (titles), Hanken Grotesk
% (body) and JetBrains Mono (paths, commands), shipped in `fonts/` under
% the SIL OFL, and the icons are Font Awesome 5 Free.
%
% The paper may be read in grayscale, so no meaning is carried by colour
% alone: every fill is paired with a shape, a border style, an icon or a
% word. Heat is used the way DESIGN.md uses it — for the hard part: the
% live path through a figure, and the point where it breaks.

\usepackage{fontspec}
\usepackage{amsmath}
\usepackage{xcolor}
\usepackage{fontawesome5}
\usepackage{tikz}
\usetikzlibrary{
  arrows.meta,
  backgrounds,
  calc,
  fit,
  positioning,
  shapes.geometric,
  shapes.misc,
}

% --- type ----------------------------------------------------------------

  BoldItalicFont=*-700]
\setsansfont{HankenGrotesk}[
  Path=fonts/, Extension=.ttf,
  UprightFont=*-400, BoldFont=*-700, ItalicFont=*-400-italic,
  BoldItalicFont=*-700]

  Path=fonts/, Extension=.ttf, Scale=0.9,
  UprightFont=*-Regular, BoldFont=*-Bold, ItalicFont=*-Italic,
  BoldItalicFont=*-Bold]
\newfontfamily\display{Quicksand}[
  Path=fonts/, Extension=.ttf,
  UprightFont=*-Medium, BoldFont=*-Bold]
\newfontfamily\semibold{HankenGrotesk-600}[Path=fonts/, Extension=.ttf]

% --- shapes --------------------------------------------------------------
\tikzset{
  % text
  every node/.style={font=\small, text=ink},
  title/.style={font=\display\bfseries\normalsize, text=ink, align=center},
  sub/.style={font=\semibold\small, text=ink, align=center},
  note/.style={font=\footnotesize, text=inksec, align=left, inner sep=1pt},
  meta/.style={font=\ttfamily\scriptsize, text=muted, align=left,
    inner sep=1pt},
  lbl/.style={font=\footnotesize, text=inksec, fill=canvas,
    inner sep=1.5pt, align=center},
  % containers. A solid border is something we own; a dashed one is a
  % boundary that was imposed on something we do not trust.
  panel/.style={draw=strong, fill=surface, rounded corners=6pt,
    inner sep=7pt, line width=0.7pt},
  host/.style={draw=ink, fill=canvas, rounded corners=6pt, inner sep=7pt,
    line width=0.9pt},
  sandbox/.style={draw=cool, fill=cool!4, rounded corners=6pt,
    dash pattern=on 4pt off 2.5pt, inner sep=7pt, line width=1pt},
  outside/.style={panel},
  % leaves
  box/.style={draw=strong, fill=canvas, rounded corners=3pt, align=left,
    inner sep=4.5pt, line width=0.7pt},
  device/.style={box, align=center},
  file/.style={draw=strong, fill=sunken, rounded corners=2pt, align=left,
    inner sep=4.5pt, line width=0.7pt},
  code/.style={file, font=\ttfamily\footnotesize},
  warn/.style={draw=warning, fill=warning!6, rounded corners=3pt,
    align=left, inner sep=4.5pt, line width=0.8pt},
  key/.style={draw=ink, fill=sunken, rounded corners=3pt, align=left,
    inner sep=4.5pt, line width=1.1pt},
  breach/.style={draw=heatred, fill=heatred!5, rounded corners=3pt,
    align=left, inner sep=4.5pt, line width=1.1pt},
  denied/.style={draw=heatred, fill=canvas, rounded corners=3pt,
    dash pattern=on 3pt off 2pt, align=center, inner sep=4.5pt,
    line width=0.9pt, text=muted},
  held/.style={draw=success, fill=success!5, rounded corners=3pt,
    align=left, inner sep=4.5pt, line width=0.8pt},
  db/.style={cylinder, shape border rotate=90, aspect=0.28, draw=strong,
    fill=canvas, align=center, minimum height=9mm, minimum width=13mm,
    inner sep=2pt, font=\footnotesize, line width=0.7pt},
  commit/.style={draw=ink!70, fill=sunken, align=center, inner sep=3pt,
    trapezium, trapezium left angle=72, trapezium right angle=108,
    font=\footnotesize},
  tag/.style={draw=strong, fill=canvas, rounded rectangle,
    font=\ttfamily\scriptsize, text=inksec, inner xsep=4pt,
    inner ysep=2pt, line width=0.6pt},
  % a numbered marker, for a figure whose items are an enumeration the
  % text refers back to (and only then).
  num/.style={circle, fill=ink, text=canvas, font=\semibold\scriptsize,
    inner sep=1.2pt, minimum size=4.2mm},
  hotnum/.style={num, fill=heatred},
  % edges. `flow` is data or control moving; `back` is a dashed one, used
  % where the two ends are not one hand-off but a shared state; `live` is
  % the one path the figure is about.
  flow/.style={-{Latex[length=1.8mm,width=1.4mm]}, draw=inksec,
    line width=0.7pt},
  back/.style={flow, dashed},
  live/.style={flow, draw=heatmid, line width=1.3pt},
  attack/.style={-{Latex[length=2.2mm,width=1.8mm]}, draw=heatred,
    line width=1.2pt},
  probe/.style={attack, dash pattern=on 3pt off 2pt, line width=0.9pt},
  link/.style={draw=inksec, dashed, line width=0.6pt},
  cross/.style={draw=heatred, line width=1.2pt},
}

% A crossed-out node: `\barred{name}`.
\newcommand{\barred}[1]{%
  \draw[cross] (#1.south west) -- (#1.north east);
  \draw[cross] (#1.north west) -- (#1.south east);}
